# The bridge's acfCleanMissileList() includes every registered ACF gun class
# whose type is "missile". This includes missile, BOMB and GBU definitions.
# WeaponIndices[] are one-based indexes into RegisteredIDs.
# StationGuidanceTypes[] selects native ACF guidance: Wire or Laser.
# Powered missiles use the BGM-71E flight baseline. BOMB/GBU classes marked
# nothrust remain gravity-driven with their original fins/drag/mass and still
# accept native Wire/Laser guidance.
# The bridge does not emulate guidance: ACF creates the selected native guidance object.
# Powered missile flight stats use BGM-71E ASM. Gravity-only bombs retain their selected ACF no-thrust flight stats.
#
# NATIVE ACF GUIDANCE:
# The E2 chip's Position vector output supplies the live target mass center
# to native ACF Wire/Laser guidance; Target:entity remains for compatibility.
# The bridge keeps the chip as missile.Launcher even while
# the loaded missile is parented to a station hologram. Wire and Laser both use
# the live E2 Position/Target outputs.
# Wire guidance keeps ACF's live target updates and wire-length limit; it can steer through a full 180-degree reversal.
# Laser keeps ACF's native tracking cone, and the bridge enforces clear brush line-of-sight. On loss it coasts along its current flight direction and can reacquire when a valid laser point returns inside the cone.
# All active rack stations share the master rack orientation; target locks do not steer them.
# The bridge uses a control-point-enabled copy of ACF Rocket Motor; original ACF PCF/addon files remain untouched.
# MaxSpeedMPH caps flight speed (0 disables); MissileScale scales the visual model and server collision mesh in XYZ once at load.
# GravityScale multiplies ACF's native gravity from 0 to 10 (1 = normal sv_gravity).
# HomingTurnScale is steering smoothing time in seconds: 0 = native ACF, higher = more gradual. No turn-angle limit.
# ProtectFriendlyMunitions = 1 blocks ACF damage (including blast damage) between bridge-created missiles/bombs; 0 disables it.
# WarheadFillerMass is ACF filler mass in kg; 0 keeps the selected round's default filler volume.
# A positive value overrides the actual ACF payload; ACF's scaled explosion effect follows FillerMass automatically.
# GuidanceStartDelay holds native ACF steering for each missile after a straight-ahead rack launch.
# Position refreshes at the ACF 66 Hz flight cadence from the target physics mass center (or bbox center without physics).
# Target search excludes map logic/brush entities, environment controllers, items, props, helicopter NPCs, and Wire entities. Other NPC classes remain eligible.

@trigger none

function void configureMissile(M:entity) {
    # Optional fuse choice is still created and configured by ACF itself.
    if (MissileFuse != "") { acfCleanMissileFuse(M, MissileFuse) }

    # Optional speed cap: 0 disables the cap; positive values are MPH.
    acfCleanMissileMaxSpeedMPH(M, MaxSpeedMPH)

    # XYZ model + physical collision scale. The server rebuilds the actual convex physics
    # mesh once here, before parenting/welding. A warning is printed if the source
    # physics mesh cannot be rebuilt; visual-only fallback is never reported as success.
    acfCleanMissileModelScale(M, MissileScale)

    # Gravity: 0 = weightless, 1 = normal sv_gravity, 10 = 10x.
    # Applies to powered missiles too; BOMB/GBU remain gravity-only.
    acfCleanMissileGravityScale(M, GravityScale)

    # Smooth the desired steering bearing without changing ACF agility or turn limits.
    # 0 = native ACF; 0.5 = moderate lag; 1 = heavier; 2 = very gradual.
    acfCleanMissileHomingTurnScale(M, HomingTurnScale)

    # Friendly protection is applied to every bridge-created round. ACF damage
    # is ignored only when both the damaged munition and damage source are
    # protected bridge-created munitions, including rounds from other racks.
    acfCleanMissileFriendlyProtection(M, ProtectFriendlyMunitions)

    # Per-missile ACF Rocket Motor controls. Defaults preserve the original PCF look.
    # SmokeScale = 0 hides smoke; 1 is stock; higher values make it larger.
    # SmokeLength = 1 is stock lifetime/trail; lower values shorten it.
    # SmokeColor uses RGB values from 0 to 255.
    # SmokeOpacity = 145 is the original bridge default.
    # FlameScale = 0 hides the flame; 1 is stock; higher values enlarge it independently.
    acfCleanMissileSmokeScale(M, SmokeScale)
    acfCleanMissileSmokeColor(M, SmokeColor)
    acfCleanMissileSmokeLength(M, SmokeLength)
    acfCleanMissileSmokeOpacity(M, SmokeOpacity)
    acfCleanMissileFlameScale(M, FlameScale)
}

# Each missile gets its own one-shot anonymous Lambda timer.
# The timer releases that missile's native ACF guidance independently of all
# other stations and later launches.
function void startGuidanceDelay(M:entity) {
    if (GuidanceStartDelay <= 0 | !M:isValid()) { return }

    # acfCleanMissileLaunchHeld() already enabled the hold before ACF's
    # synchronous first flight Think. This Lambda only releases that hold.
    timer(GuidanceStartDelay, function() {
        if (M:isValid()) {
            acfCleanMissileGuidanceHold(M, 0)
        }
    })
}

function void loadStation(S:number) {
    if (S < 1 | S > MaxRockets) { return }
    if (StationEnabled[S, number] == 0 | StationLoaded[S, number] != 0) { return }
    WorkHolo = holoEntity(9000 + S)
    if (!WorkHolo:isValid()) {
        Status = "LOAD FAILED: station hologram " + S + " is invalid"
        return
    }

    WorkMissile = acfCleanMissileCreateGuidedFiller(WeaponIndices[S, number], WorkHolo:pos(), WorkHolo:angles(), StationGuidanceTypes[S, string], WarheadFillerMass)
    if (!WorkMissile:isValid()) {
        Status = "LOAD FAILED station " + S + ": " + acfCleanMissileStatus()
        CreateStatus = acfCleanMissileStatus()
        return
    }

    # Apply configuration (including the one-time XYZ collision rebuild) before
    # parenting the missile to its rack hologram.
    configureMissile(WorkMissile)
    acfCleanMissileParent(WorkMissile, WorkHolo)
    # Loaded weapons occupy the rack but must not collide with other entities.
    # Collision group 5 is set in the Lua bridge after SetParent() so E2
    # doesn't need to call it here.
    WorkMissile:noCollideAll(1)
    Missiles[S, entity] = WorkMissile
    StationLoaded[S, number] = 1
    findExcludeEntity(WorkMissile)
    Status = "STATION " + S + " LOADED"
    CreateStatus = acfCleanMissileStatus()
}

function void loadAllStations() {
    for (I = 1, MaxRockets) {
        loadStation(I)
    }
}

function void serviceStationReloads() {
    # Check every station for an unexpectedly lost round, independently of
    # the normal post-launch reload timer.
    # This server's E2 build lacks the hidden-state accessor, so detonated rounds
    # are detected when ACF finishes removing the entity and isValid() becomes 0.
    for (I = 1, MaxRockets) {
        if (StationEnabled[I, number] != 0 & StationLoaded[I, number] != 0) {
            WorkMissile = Missiles[I, entity]
            WorkHolo = holoEntity(9000 + I)

            # A missing entity or detached round means this station is no
            # longer physically loaded and must recover now.
            if (!WorkMissile:isValid()) {
                StationLoaded[I, number] = 0
                Missiles[I, entity] = noentity()
                StationReloading[I, number] = 1
                StationReloadAt[I, number] = curtime()
                Status = "STATION " + I + " ROUND LOST - RELOADING"
            } elseif (WorkHolo:isValid() & WorkMissile:parent() != WorkHolo) {
                # A live round that has become detached is not a valid rack load.
                acfCleanMissileRemove(WorkMissile)
                StationLoaded[I, number] = 0
                Missiles[I, entity] = noentity()
                StationReloading[I, number] = 1
                StationReloadAt[I, number] = curtime()
                Status = "STATION " + I + " ROUND DETACHED - RELOADING"
            }
        }

        # Normal timed reloads and immediate recovery reloads share this
        # per-station service. Immediate recovery sets the deadline to now.
        if (StationReloading[I, number] != 0 & curtime() >= StationReloadAt[I, number]) {
            if (StationEnabled[I, number] != 0 & StationLoaded[I, number] == 0) {
                loadStation(I)
                if (StationLoaded[I, number] != 0) {
                    StationReloading[I, number] = 0
                    StationReloadAt[I, number] = 0
                } else {
                    StationReloadAt[I, number] = curtime() + 1
                }
            } else {
                StationReloading[I, number] = 0
                StationReloadAt[I, number] = 0
            }
        }
    }
}

function number fireStation(S:number, AutoMode:number) {
    if (S < 1 | S > MaxRockets) { return 0 }
    if (StationLoaded[S, number] == 0) { return 0 }
    WorkMissile = Missiles[S, entity]
    if (!WorkMissile:isValid()) {
        StationLoaded[S, number] = 0
        Missiles[S, entity] = noentity()
        return 0
    }

    WorkTarget = Targets[S, entity]
    if (AutoMode != 0 & !WorkTarget:isValid()) { return 0 }
    if (AutoMode != 0 & PreventRepeatPerLock != 0 & WorkTarget:isValid() & StationFired[S, number] != 0) { return 0 }

    WorkHolo = holoEntity(9000 + S)
    if (!WorkHolo:isValid()) { return 0 }

    # SAM-style launch: leave the rack along the station's fixed forward axis.
    # The locked target is deliberately NOT used for initial launch direction.
    # Native ACF guidance is held for GuidanceStartDelay, then released by an
    # independent one-shot Lambda timer for this exact missile.
    WorkDirection = WorkHolo:forward()

    # Re-enable normal collision as the weapon leaves the rack.
    WorkMissile:noCollideAll(0)
    if (acfCleanMissileLaunchHeld(WorkMissile, WorkDirection, GuidanceStartDelay > 0) != 0) {
        if (GuidanceStartDelay > 0) { startGuidanceDelay(WorkMissile) }
        Missiles[S, entity] = noentity()
        StationLoaded[S, number] = 0
        StationFired[S, number] = 1
        # Independent reload: this station's timer starts at its successful launch.
        StationReloading[S, number] = 1
        StationReloadAt[S, number] = curtime() + ReloadDelay
        Status = "STATION " + S + " LAUNCHED"
        return 1
    }

    # Launch failed; keep the loaded weapon non-colliding inside the rack.
    WorkMissile:noCollideAll(1)
    Status = "LAUNCH FAILED station " + S + ": " + acfCleanMissileStatus()
    return 0
}

function void launchQueueStep() {
    if (LaunchQueueActive == 0) { return }

    # Stop as soon as the configured number of SUCCESSFUL launches is reached.
    # MissilesPerClick = 0 is the special single-shot setting.
    if (LaunchQueueFired >= LaunchQueueShotLimit) {
        LaunchQueueActive = 0
        stoptimer("launch_queue")
        Status = "LAUNCH QUEUE COMPLETE - " + LaunchQueueFired + " STATION(S) FIRED"
        if (DebugTargets != 0) { print("RACK QUEUE FINISHED", "shots", LaunchQueueFired, "limit", LaunchQueueShotLimit, "mode", LaunchQueueAutoMode) }
        return
    }

    # Process ONLY one station per timer pass. ACF may reject a launch while
    # its weapon/launcher is still on cooldown; never burn through remaining
    # stations in the same E2 execution after one rejected launch.
    if (LaunchQueueNext > MaxRockets) {
        LaunchQueueActive = 0
        stoptimer("launch_queue")
        if (DebugTargets != 0) { print("RACK QUEUE FINISHED", "shots", LaunchQueueFired, "mode", LaunchQueueAutoMode) }
        if (LaunchQueueFired == 0 & LaunchQueueAutoMode == 0) {
            Status = "FIRE BLOCKED: NO ELIGIBLE LOADED STATIONS"
        } elseif (LaunchQueueFired > 0) {
            Status = "LAUNCH QUEUE COMPLETE - " + LaunchQueueFired + " STATION(S) FIRED"
        }
        return
    }

    WorkStation = LaunchQueueNext
    WorkTarget = Targets[WorkStation, entity]

    if (StationEnabled[WorkStation, number] == 0) {
        if (DebugTargets != 0) { print("RACK QUEUE SKIP", WorkStation, "disabled") }
        LaunchQueueNext = WorkStation + 1
        LaunchQueueRetryCount = 0
        return
    }

    if (StationLoaded[WorkStation, number] == 0) {
        if (DebugTargets != 0) { print("RACK QUEUE WAIT", WorkStation, "not loaded; reload pending", StationReloading[WorkStation, number]) }
        # Do not skip a station that is still reloading. Wait for its own
        # reload service to make it ready, preserving strict front-to-rear order.
        if (StationReloading[WorkStation, number] != 0) {
            return
        }
        # No reload is pending, so this station cannot currently participate.
        LaunchQueueNext = WorkStation + 1
        LaunchQueueRetryCount = 0
        return
    }

    if (DebugTargets != 0) {
        if (WorkTarget:isValid()) {
            print("RACK QUEUE TRY", WorkStation, "attempt", LaunchQueueRetryCount + 1, "loaded", StationLoaded[WorkStation, number], "target", WorkTarget:id(), WorkTarget:getClass(), WorkTarget:name(), WorkTarget:model())
        } else {
            print("RACK QUEUE TRY", WorkStation, "attempt", LaunchQueueRetryCount + 1, "loaded", StationLoaded[WorkStation, number], "target NONE")
        }
    }

    WorkFired = fireStation(WorkStation, LaunchQueueAutoMode)
    if (DebugTargets != 0) { print("RACK QUEUE RESULT", WorkStation, WorkFired, Status) }

if (WorkFired != 0) {
    LaunchQueueFired++
    SalvoShots = LaunchQueueFired
    LaunchQueueNext = (WorkStation % MaxRockets) + 1
    LaunchQueueRetryCount = 0


        if (LaunchQueueFired >= LaunchQueueShotLimit) {
            LaunchQueueActive = 0
            stoptimer("launch_queue")
            Status = "LAUNCH QUEUE COMPLETE - " + LaunchQueueFired + " STATION(S) FIRED"
            if (DebugTargets != 0) { print("RACK QUEUE FINISHED", "shots", LaunchQueueFired, "limit", LaunchQueueShotLimit, "mode", LaunchQueueAutoMode) }
            return
        }
    } else {
        # Retry this station on a later timer pass instead of immediately
        # trying every remaining missile in one execution. After the retry
        # limit, advance so one faulty station cannot deadlock the rack.
        LaunchQueueRetryCount++
        if (LaunchQueueRetryCount > LaunchQueueRetryLimit) {
            if (DebugTargets != 0) { print("RACK QUEUE SKIP", WorkStation, "retry limit reached", LaunchQueueRetryLimit, Status) }
            LaunchQueueNext = WorkStation + 1
            LaunchQueueRetryCount = 0
        }
    }

}

function void launchSalvo(AutoMode:number) {
    if (LaunchQueueActive != 0) { return }

LaunchQueueActive = 1
# Keep the persistent station pointer, but wrap it if MaxRockets was reduced.
if (LaunchQueueNext < 1 | LaunchQueueNext > MaxRockets) { LaunchQueueNext = 1 }
LaunchQueueAutoMode = AutoMode
LaunchQueueFired = 0
LaunchQueueRetryCount = 0
    
    
    if (MissilesPerClick > 0) {
        LaunchQueueShotLimit = min(MissilesPerClick, MaxRockets)
    } else {
        LaunchQueueShotLimit = 1
    }
    SalvoShots = 0
    Status = "LAUNCH QUEUE STARTED"
    if (DebugTargets != 0) { print("RACK QUEUE STARTED", "mode", AutoMode, "loaded", LoadedCount) }
    # One repeating callback owns the queue lifecycle. Do not recreate a
    # same-named callback timer from inside its own callback: Wiremod keeps
    # it registered until that callback returns.
    timer("launch_queue", LaunchInterval, 0, function() { launchQueueStep() })
    launchQueueStep()
}

# ACF native Wire guidance prefers the live Position vector over Target:entity.
# E2 owns the aim point: use the real physics mass center whenever the target
# has valid physics; non-physics entities use their world-space bounding-box center.
# Refresh at 66 Hz to match ACF missile CalcFlight(), not a 20 Hz stepped target.
function void updateGuidancePosition() {
    if (!Target:isValid()) {
        Position = vec(0)
        return
    }

    if (Target:isValidPhysics()) {
        Position = Target:massCenter()
    } else {
        Position = Target:boxCenterW()
    }
}

function void scanTargets() {
    SearchHolo = holoEntity(9100)
    if (!SearchHolo:isValid()) { return }
    if (!findCanQuery()) { return }

    WorkFindCount = findInCone(SearchHolo:pos(), SearchHolo:forward(), FindingDistance, FindingConeAngle / 2)

    # Diagnostic output: print the actual entities returned by findInCone,
    # including class, map name, model, entity index, and position.
    # Rate-limited so a 0.25-second scan does not flood chat.
    if (DebugTargets != 0 & curtime() >= LastDebugPrint + DebugPrintInterval) {
        LastDebugPrint = curtime()
        print("RACK FIND DEBUG", "count", WorkFindCount, "range", FindingDistance, "cone", FindingConeAngle)
        for (J = 1, min(WorkFindCount, 16)) {
            WorkCandidate = findResult(J)
            if (WorkCandidate:isValid()) {
                print("RACK FOUND", J, "ent", WorkCandidate:id(), "class", WorkCandidate:getClass(), "name", WorkCandidate:name(), "model", WorkCandidate:model(), "pos", WorkCandidate:pos())
            }
        }
        if (WorkTarget:isValid()) {
            print("RACK TARGET", "ent", WorkTarget:id(), "class", WorkTarget:getClass(), "name", WorkTarget:name(), "model", WorkTarget:model())
        } else {
            print("RACK TARGET", "none")
        }
    }

    if (MultiTargetEnabled == 0) {
        WorkTarget = Targets[1, entity]
        WorkPresent = 0
        if (WorkTarget:isValid()) {
            for (J = 1, WorkFindCount) {
                if (findResult(J) == WorkTarget) { WorkPresent = 1 }
            }
        }
        if (!WorkTarget:isValid() | WorkPresent == 0) {
            if (WorkFindCount > 0) {
                WorkTarget = findResult(1)
            } else {
                WorkTarget = noentity()
            }
            if (WorkTarget != Targets[1, entity]) {
                for (I = 1, MaxRockets) { StationFired[I, number] = 0 }
            }
        }
        for (I = 1, MaxRockets) { Targets[I, entity] = WorkTarget }
        Target = WorkTarget
    } else {
        for (I = 1, MaxRockets) {
            WorkTarget = Targets[I, entity]
            WorkPresent = 0
            if (WorkTarget:isValid()) {
                for (J = 1, WorkFindCount) {
                    if (findResult(J) == WorkTarget) { WorkPresent = 1 }
                }
            }
            if (!WorkTarget:isValid() | WorkPresent == 0) {
                Targets[I, entity] = noentity()
                StationFired[I, number] = 0
            }
        }

        if (WorkTarget:isValid()) { Target = WorkTarget } else { Target = noentity() }

        for (I = 1, MaxRockets) {
            WorkTarget = Targets[I, entity]
            if (StationEnabled[I, number] != 0 & !WorkTarget:isValid()) {
                for (J = 1, WorkFindCount) {
                    WorkCandidate = findResult(J)
                    WorkAssigned = 0
                    for (K = 1, MaxRockets) {
                        if (K != I & Targets[K, entity] == WorkCandidate) { WorkAssigned = 1 }
                    }
                    if (WorkAssigned == 0) {
                        Targets[I, entity] = WorkCandidate
                        StationFired[I, number] = 0
                        break
                    }
                }
            }
        }
    }

    TargetCount = 0
    if (MultiTargetEnabled == 0) {
        WorkTarget = Targets[1, entity]
        if (WorkTarget:isValid()) { TargetCount = 1 }
    } else {
        for (I = 1, MaxRockets) {
            WorkTarget = Targets[I, entity]
            if (WorkTarget:isValid()) { TargetCount = TargetCount + 1 }
        }
    }

    updateGuidancePosition()

    if (DebugTargets != 0 & curtime() == LastDebugPrint) {
        for (I = 1, MaxRockets) {
            WorkTarget = Targets[I, entity]
            if (WorkTarget:isValid()) {
                print("RACK ASSIGN", "station", I, "ent", WorkTarget:id(), "class", WorkTarget:getClass(), "name", WorkTarget:name())
            }
        }
    }

    if (AutoLaunchEnabled != 0 & LaunchQueueActive == 0) {
        WorkReady = 0
        for (I = 1, MaxRockets) {
            WorkTarget = Targets[I, entity]
            if (StationLoaded[I, number] != 0 & WorkTarget:isValid() & (PreventRepeatPerLock == 0 | StationFired[I, number] == 0)) {
                WorkReady = 1
            }
        }
        if (WorkReady != 0) { launchSalvo(1) }
    }
}

function void updateLoadedCount() {
    LoadedCount = 0
    for (I = 1, MaxRockets) {
        WorkMissile = Missiles[I, entity]
        if (StationLoaded[I, number] != 0 & WorkMissile:isValid()) { LoadedCount = LoadedCount + 1 }
    }
    Reloading = 0
    for (I = 1, MaxRockets) {
        if (StationReloading[I, number] != 0) { Reloading++ }
    }
}

event input(InputName:string) {
    # Rising-edge detection: one press creates one activation; holding Fire does not repeat.
    if (InputName == "Fire" & Fire != 0 & LastFire == 0) {
        launchSalvo(0)
    }
    if (InputName == "Reset" & Reset != 0 & LastReset == 0) {
        if (SetupComplete == 0) {
            Status = "RESET BLOCKED: STATION HOLOGRAMS ARE STILL BUILDING"
        } else {
            stoptimer("launch_queue")
            LaunchQueueActive = 0
            LaunchQueueNext = 1
            LaunchQueueFired = 0
            for (I = 1, MaxRockets) {
                StationReloading[I, number] = 0
                StationReloadAt[I, number] = 0
                WorkMissile = Missiles[I, entity]
                if (WorkMissile:isValid()) { acfCleanMissileRemove(WorkMissile) }
                Missiles[I, entity] = noentity()
                StationLoaded[I, number] = 0
                StationFired[I, number] = 0
                Targets[I, entity] = noentity()
            }
            loadAllStations()
            Status = "RESET: ACTIVE STATIONS RELOADING"
        }
    }
    if (InputName == "Fire") { LastFire = Fire }
    if (InputName == "Reset") { LastReset = Reset }
}

event removed(Resetting:number) {
    stopAllTimers()
    for (I = 1, 12) {
        WorkMissile = Missiles[I, entity]
        if (WorkMissile:isValid()) { acfCleanMissileRemove(WorkMissile) }
        WorkHolo = holoEntity(9000 + I)
        if (WorkHolo:isValid()) { holoDelete(9000 + I) }
    }
    SearchHolo = holoEntity(9100)
    if (SearchHolo:isValid()) { holoDelete(9100) }
    RackRootHolo = holoEntity(8999)
    if (RackRootHolo:isValid()) { holoDelete(8999) }
}
